    95→function pickRandom<T>(arr: T[]): T {
    96→  return arr[Math.floor(Math.random() * arr.length)]
    97→}
    98→
    99→function shuffle<T>(arr: T[]): T[] {
   100→  const a = [...arr]
   101→  for (let i = a.length - 1; i > 0; i--) {
   102→    const j = Math.floor(Math.random() * (i + 1))
   103→    ;[a[i], a[j]] = [a[j], a[i]]
   104→  }
   105→  return a
   106→}

   434→      const dy = e.clientY - cy

   436→      const strength = Math.min(dist / 220, 1)
   437→      pupilX.set((dx / dist) * 2.4 * strength)
   438→      pupilY.set((dy / dist) * 2 * strength)
   439→    }
   440→    window.addEventListener('mousemove', onMouseMove, { passive: true })
   441→    return () => window.removeEventListener('mousemove', onMouseMove)

   502→    setPos({
   503→      x: clamp(e.clientX - d.offX, minX - 6, maxX + 10),
   504→      y: clamp((window.innerHeight - e.clientY) - d.offY, 10, Math.max(10, vp.h - mascotSize - 50)),

   977→              />
   978→              <button
   979→                type="submit"
   980→                disabled={thinking || !input.trim() || guideRunning}
   981→                className="btn-gradient-primary flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-white disabled:opacity-50 transition"
   982→                aria-label="Enviar"

   984→                <Send className="h-3 w-3" />

   986→            </form>
   987→          </motion.div>
   988→        )}
   989→      </AnimatePresence>
   990→
   991→      {/* ===== El muñequito (caminante y arrastrable) ===== */}
   992→      <motion.button
   993→        ref={btnRef}
   994→        onClick={() => { if (wasDraggedRef.current) { wasDraggedRef.current = false; return } }}
   995→        onDoubleClick={() => minimizePixel()}
   996→        onPointerDown={onPointerDown}
   997→        onPointerMove={onPointerMove}
   998→        onPointerUp={onPointerUp}
   999→        onPointerCancel={onPointerCancel}
  1000→        animate={{ x: pos.x, y: -pos.y }}
  1001→        transition={dragging ? { duration: 0 } : walking ? { duration: walkDur, ease: 'linear' } : { type: 'spring', stiffness: 60, damping: 15 }}
  1002→        onAnimationComplete={() => {
  1003→          setWalking((w) => {
  1004→            if (w) savePos(pos)
  1005→            return false
  1006→          })
  1007→        }}
  1008→        className={cn(
  1009→          'pointer-events-auto absolute left-0 bottom-0 cursor-grab select-none touch-none',
  1010→          dragging && 'cursor-grabbing z-30'
  1011→        )}
  1012→        title="Pixel · arrástrame donde quieras, tócame para hablar (doble clic minimiza)"
  1013→        aria-label="Pixel, asistente de IA"
  1014→      >
  1015→        {/* Flotación suave (solo cuando está quieto) */}
  1016→        <motion.div
  1017→          animate={walking || dragging ? { y: 0 } : mood === 'sleepy' ? { y: [0, -2, 0] } : { y: [0, -5, 0] }}
  1018→          transition={{ duration: mood === 'sleepy' ? 4 : 2.4, repeat: walking || dragging ? 0 : Infinity, ease: 'easeInOut' }}
  1019→          whileHover={{ scale: walking || dragging ? 1 : 1.1 }}
  1020→        >
  1021→          {/* Emotes de un disparo (salto / baile) */}
  1022→          <motion.div
  1023→            key={emote.key}
  1024→            animate={
  1025→              dragging ? { rotate: [-4, 4, -4] }
  1026→                : emote.kind === 'jump'
  1027→                  ? { y: [0, -20, 0, -8, 0], rotate: 0 }
  1028→                  : emote.kind === 'dance'
  1029→                    ? { rotate: [0, -8, 8, -6, 6, 0], y: [0, -3, 0, -3, 0, 0] }
  1030→                    : { y: 0, rotate: 0 }
  1031→            }
  1032→            transition={dragging ? { duration: 0.45, repeat: Infinity, ease: 'easeInOut' } : { duration: emote.kind === 'jump' ? 0.9 : 1.4, ease: 'easeInOut' }}
  1033→          >
  1034→            <motion.div animate={{ scaleX: facing }} transition={{ duration: 0.25 }}>
  1035→              <PixelIcon size={mascotSize} pupilX={pupilX} pupilY={pupilY} mood={mood} walking={walking || dragging} />
  1036→            </motion.div>

  1041→          </motion.div>
  1042→        </motion.div>
  1043→      </motion.button>
  1044→    </div>
  1045→  )
  1046→}
