   379→    queryKey: ['user-profile', userId],
   380→    queryFn: () => userService.get(userId),
   381→    enabled: !isMe,
   382→  })
   383→

   385→  const u = profileData?.user
   386→
   387→  const cards = isMe && stats ? [
   388→    { icon: FileText, label: 'Posts', value: stats.posts, gradient: 'from-wine-400 to-wine-500' },

   391→    { icon: Heart, label: 'Likes', value: stats.totalLikesReceived, gradient: 'from-wine-500 to-wine-700' },
   392→    { icon: Gamepad2, label: 'Betas', value: stats.betas, gradient: 'from-olive-400 to-sepia-500' },
   393→    { icon: Star, label: 'Nivel', value: stats.level, gradient: 'from-bronze-400 to-wine-500' },
   394→  ] : u ? [

   396→    { icon: Users, label: 'Seguidores', value: u.followersCount, gradient: 'from-wine-400 to-wine-500' },
   397→    { icon: Users, label: 'Siguiendo', value: u.followingCount, gradient: 'from-olive-400 to-sepia-500' },
   398→    { icon: Gamepad2, label: 'Betas', value: 0, gradient: 'from-amber-400 to-bronze-500' },
   399→  ] : []
   400→
   401→  if (isLoading) {
   402→    return (
   403→      <div className="grid grid-cols-3 md:grid-cols-6 gap-2">
   404→        {[1,2,3,4,5,6].map(i => <Skeleton key={i} className="h-20 rounded-md" />)}

   408→

   422→              <Icon className="h-4 w-4" />
   423→            </div>
   424→            <AnimatedCounter value={card.value} />
   425→            <div className="text-[10px] text-muted-foreground">{card.label}</div>

   430→  )

   433→function AnimatedCounter({ value }: { value: number }) {
   434→  const [display, setDisplay] = useState(0)
   435→  useEffect(() => {
   436→    const duration = 800
   437→    const start = Date.now()
   438→    const interval = setInterval(() => {
   439→      const elapsed = Date.now() - start
   440→      const progress = Math.min(elapsed / duration, 1)
   441→      setDisplay(Math.floor(value * progress))
   442→      if (progress >= 1) clearInterval(interval)
   443→    }, 16)
   444→    return () => clearInterval(interval)
   445→  }, [value])
   446→  return <div className="text-lg font-bold">{display}</div>
   447→}

   998→// ===== EDIT PROFILE DIALOG (completo) =====
   999→function EditProfileDialog({ open, onClose, user, onSaved }: any) {
  1000→  const { refresh } = useCurrentUser()
  1001→  const [bio, setBio] = useState(user.bio ?? '')
  1002→  const [avatar, setAvatar] = useState(user.avatar ?? '')
  1003→  const [banner, setBanner] = useState(user.banner ?? '/uploads/default-banner.png')
  1004→  const [fullName, setFullName] = useState(user.fullName ?? '')
  1005→  const [location, setLocation] = useState(user.location ?? '')
  1006→  const [website, setWebsite] = useState(user.website ?? '')
  1007→  const [profession, setProfession] = useState(user.profession ?? '')
  1008→  const [birthDate, setBirthDate] = useState(user.birthDate ? user.birthDate.split('T')[0] : '')
  1009→  const [socialLinks, setSocialLinks] = useState<SocialLinks>(user.socialLinks || {})
  1010→  const [selectedTags, setSelectedTags] = useState<string[]>(user.tags ?? [])
  1011→  const [uploadingAvatar, setUploadingAvatar] = useState(false)
  1012→  const [uploadingBanner, setUploadingBanner] = useState(false)
  1013→  const [saving, setSaving] = useState(false)
  1014→
  1015→  function toggleTag(id: string) {
  1016→    setSelectedTags((prev) =>
  1017→      prev.includes(id) ? prev.filter(t => t !== id) : prev.length < 8 ? [...prev, id] : prev
  1018→    )
  1019→  }
  1020→
  1021→  function updateSocial(key: keyof SocialLinks, value: string) {
  1022→    setSocialLinks((prev) => ({ ...prev, [key]: value || undefined }))
  1023→  }
  1024→
  1025→  async function handleUpload(e: React.ChangeEvent<HTMLInputElement>, type: 'avatar' | 'banner') {
  1026→    const f = e.target.files?.[0]
  1027→    if (!f) return
  1028→    if (type === 'avatar') setUploadingAvatar(true)
  1029→    else setUploadingBanner(true)
  1030→    try {
  1031→      const res = await uploadService.upload(f, 'avatar')
  1032→      if (type === 'avatar') setAvatar(res.url)
  1033→      else setBanner(res.url)
  1034→      toast.success(type === 'avatar' ? 'Foto de perfil actualizada' : 'Portada actualizada')
  1035→    } catch (err: any) {
  1036→      toast.error(err.message || 'Error')
  1037→    } finally {
  1038→      if (type === 'avatar') setUploadingAvatar(false)
  1039→      else setUploadingBanner(false)
  1040→      e.target.value = ''
  1041→    }
  1042→  }
  1043→
  1044→  async function handleSave() {
  1045→    setSaving(true)
  1046→    try {
  1047→      await userService.updateProfile({
  1048→        bio: bio.trim() || null,
  1049→        avatar: avatar || null,
  1050→        banner: banner || null,
  1051→        fullName: fullName.trim() || null,
  1052→        location: location.trim() || null,
  1053→        website: website.trim() || null,
  1054→        profession: profession.trim() || null,
  1055→        birthDate: birthDate || null,
  1056→        socialLinks: Object.keys(socialLinks).length > 0 ? socialLinks : null,
  1057→        tags: selectedTags,
  1058→      })
  1059→      await refresh()
  1060→      toast.success('Perfil actualizado')
  1061→      onSaved()
  1062→    } catch (err: any) {
  1063→      toast.error(err.message || 'Error')
  1064→    } finally {
  1065→      setSaving(false)
  1066→    }
  1067→  }
  1068→
  1069→  if (!open) return null
  1070→
  1071→  return (
  1072→    <motion.div
  1073→      initial={{ opacity: 0 }}
  1074→      animate={{ opacity: 1 }}
  1075→      exit={{ opacity: 0 }}
  1076→      className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 p-4"
  1077→      onClick={onClose}
  1078→    >
  1079→      <motion.div
  1080→        initial={{ scale: 0.95 }}
  1081→        animate={{ scale: 1 }}
  1082→        exit={{ scale: 0.95 }}
  1083→        className="glass-strong w-full max-w-lg max-h-[90vh] overflow-y-auto custom-scroll rounded-lg"
  1084→        onClick={(e) => e.stopPropagation()}
  1085→      >
  1086→        {/* Header con preview de portada + avatar */}
  1087→        <div className="relative">
  1088→          {/* Banner */}
  1089→          <div className="h-28 relative group overflow-hidden rounded-t-2xl">
  1090→            {banner ? (
  1091→              <img src={banner} alt="" className="absolute inset-0 w-full h-full object-cover" />
  1092→            ) : (
  1093→              <div className="absolute inset-0 w-full h-full banner-gradient" />
  1094→            )}
  1095→            <label className="absolute top-2 right-2 z-20 cursor-pointer rounded-full bg-black/50 backdrop-blur p-2 text-white hover:bg-black/70 transition">
  1096→              <input type="file" accept="image/*" className="hidden" onChange={(e) => handleUpload(e, 'banner')} />
