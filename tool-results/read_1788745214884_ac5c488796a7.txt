   378→function ProfileStats({ userId }: { userId: string }) {
   379→  const isMe = useCurrentUser().user?.id === userId
   380→  const { data, isLoading } = useQuery({

   384→  })
   385→
   386→  const { data: profileData } = useQuery({
   387→    queryKey: ['user-profile', userId],
   388→    queryFn: () => userService.get(userId),
   389→    enabled: !isMe,
   390→  })
   391→
   392→  const stats = isMe ? data?.stats : null
   393→  const u = profileData?.user
   394→
   395→  const rows = isMe && stats ? [
   396→    { icon: FileText, label: 'Posts', value: stats.posts, gradient: 'from-wine-400 to-wine-500' },
   397→    { icon: Users, label: 'Seguidores', value: stats.followers, gradient: 'from-wine-500 to-wine-700' },
   398→    { icon: Users, label: 'Siguiendo', value: stats.following, gradient: 'from-olive-400 to-sepia-500' },
   399→    { icon: Gamepad2, label: 'Betas', value: stats.betas, gradient: 'from-amber-400 to-bronze-500' },
   400→    { icon: Download, label: 'Descargas', value: stats.totalDownloads, gradient: 'from-bronze-400 to-bronze-600' },
   401→    { icon: Heart, label: 'Likes recibidos', value: stats.totalLikesReceived, gradient: 'from-wine-400 to-sepia-500' },
   402→  ] : u ? [
   403→    { icon: FileText, label: 'Posts', value: u.postsCount, gradient: 'from-wine-400 to-wine-500' },
   404→    { icon: Users, label: 'Seguidores', value: u.followersCount, gradient: 'from-wine-500 to-wine-700' },
   405→    { icon: Users, label: 'Siguiendo', value: u.followingCount, gradient: 'from-olive-400 to-sepia-500' },
   406→  ] : []
   407→
   408→  if (isLoading) {
   409→    return (
   410→      <div className="glass-card p-4">
   411→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   412→          {[1,2,3,4,5,6].map(i => <Skeleton key={i} className="h-12 rounded-md" />)}
   413→        </div>
   414→      </div>
   415→    )
   416→  }
   417→
   418→  if (rows.length === 0) return null
   419→
   420→  return (
   421→    <div className="glass-card p-4">
   422→      <h3 className="font-bold text-sm mb-3 flex items-center gap-1.5">
   423→        <Activity className="h-4 w-4 text-primary" />

   425→      </h3>

   431→              <div className={cn('flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-gradient-to-br text-white shadow-sm', row.gradient)}>
   432→                <Icon className="h-3.5 w-3.5" />
   433→              </div>
   434→              <div className="min-w-0">
   435→                <p className="text-sm font-bold leading-none">{row.value}</p>

   438→            </div>

   442→    </div>

   446→function AnimatedCounter({ value }: { value: number }) {
   447→  const [display, setDisplay] = useState(0)
   448→  useEffect(() => {
   449→    const duration = 800
   450→    const start = Date.now()
   451→    const interval = setInterval(() => {
   452→      const elapsed = Date.now() - start
   453→      const progress = Math.min(elapsed / duration, 1)
   454→      setDisplay(Math.floor(value * progress))
   455→      if (progress >= 1) clearInterval(interval)
   456→    }, 16)
   457→    return () => clearInterval(interval)
   458→  }, [value])
   459→  return <div className="text-lg font-bold">{display}</div>
   460→}

  1030→  function toggleTag(id: string) {
  1031→    setSelectedTags((prev) =>

  1033→    )

  1035→
  1036→  function updateSocial(key: keyof SocialLinks, value: string) {
  1037→    setSocialLinks((prev) => ({ ...prev, [key]: value || undefined }))

  1039→
  1040→  async function handleUpload(e: React.ChangeEvent<HTMLInputElement>, type: 'avatar' | 'banner') {
  1041→    const f = e.target.files?.[0]
  1042→    if (!f) return
  1043→    if (type === 'avatar') setUploadingAvatar(true)
  1044→    else setUploadingBanner(true)
  1045→    try {
  1046→      const res = await uploadService.upload(f, 'avatar')
  1047→      if (type === 'avatar') setAvatar(res.url)
  1048→      else setBanner(res.url)
  1049→      toast.success(type === 'avatar' ? 'Foto de perfil actualizada' : 'Portada actualizada')
  1050→    } catch (err: any) {
  1051→      toast.error(err.message || 'Error')
  1052→    } finally {
  1053→      if (type === 'avatar') setUploadingAvatar(false)
  1054→      else setUploadingBanner(false)
  1055→      e.target.value = ''
  1056→    }
  1057→  }

  1070→        birthDate: birthDate || null,
  1071→        socialLinks: Object.keys(socialLinks).length > 0 ? socialLinks : null,
  1072→        tags: selectedTags,
  1073→      })
  1074→      await refresh()
  1075→      toast.success('Perfil actualizado')
  1076→      onSaved()
  1077→    } catch (err: any) {
  1078→      toast.error(err.message || 'Error')
  1079→    } finally {

  1081→    }
  1082→  }
  1083→
  1084→  if (!open) return null
  1085→
  1086→  return (
  1087→    <motion.div
  1088→      initial={{ opacity: 0 }}
