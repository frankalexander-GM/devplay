    96→    96→  function getPasswordStrength(pwd: string): { score: number; label: string; color: string } {
    97→    97→    let score = 0
    98→    98→    if (pwd.length >= 6) score++

   100→   100→    if (/[A-Z]/.test(pwd) && /[a-z]/.test(pwd)) score++

   104→   104→    const colors = ['bg-red-500', 'bg-bronze-500', 'bg-amber-500', 'bg-lime-500', 'bg-olive-500', 'bg-olive-600']
   105→   105→    return { score, label: labels[score] || 'Muy débil', color: colors[score] || 'bg-red-500' }
   106→   106→  }
