   378→  function onPointerDown(e: React.PointerEvent) {
   379→    if (collapsed) return

   385→    dragRef.current = { startX: e.clientX, startY: e.clientY, offX: e.clientX - p.x, offY: (window.innerHeight - e.clientY) - p.y, moved: false }
   386→  }

   388→  function onPointerMove(e: React.PointerEvent) {
   389→    const d = dragRef.current
   390→    if (!d) return
   391→    const dx = e.clientX - d.startX
   392→    const dy = (window.innerHeight - e.clientY) - (window.innerHeight - d.startY)
   393→    if (!d.moved && Math.hypot(dx, dy) > 7) {
   394→      d.moved = true
   395→      wasDraggedRef.current = true
   396→      setDragging(true)
   397→      setBubbleVisible(false)
   398→      setChatOpen(false)
   399→      setRegisterOpen(false)
   400→      setMood('excited')
   401→    }
   402→    if (!d.moved) return
   403→    const { minX, maxX } = limits()
   404→    setFacing(dx >= 0 ? 1 : -1)
   405→    setPos({
   406→      x: clamp(e.clientX - d.offX, minX - 6, maxX + 10),
   407→      y: clamp((window.innerHeight - e.clientY) - d.offY, floorY, Math.max(floorY, visibleH - mascotSize - 14)),
   408→    })
   409→  }

   412→    const d = dragRef.current

   416→      wasDraggedRef.current = false
   417→      setDragging(false)
   418→      startFall()
   419→    } else {
   420→      handleBuddyClick()
   421→    }
   422→  }
