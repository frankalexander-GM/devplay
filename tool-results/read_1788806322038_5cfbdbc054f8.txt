    97→    97→  function getPasswordStrength(pwd: string): { score: number; label: string; color: string } {
    98→    98→    let score = 0
    99→    99→    if (pwd.length >= 6) score++
   100→   100→    if (pwd.length >= 10) score++
   101→   101→    if (/[A-Z]/.test(pwd) && /[a-z]/.test(pwd)) score++
   102→   102→    if (/[0-9]/.test(pwd)) score++
   103→   103→    if (/[^A-Za-z0-9]/.test(pwd)) score++
   104→   104→    const labels = ['Muy débil', 'Débil', 'Regular', 'Buena', 'Fuerte', 'Muy fuerte']
   105→   105→    const colors = ['bg-red-500', 'bg-bronze-500', 'bg-amber-500', 'bg-lime-500', 'bg-olive-500', 'bg-olive-600']
   106→   106→    return { score, label: labels[score] || 'Muy débil', color: colors[score] || 'bg-red-500' }
   107→   107→  }

   133→   133→        method: 'POST',

   148→   148→      }
   149→   149→    } catch {
   150→   150→      setErrors([{ field: 'loginEmail', message: 'Error de conexión. Intenta de nuevo.' }])
