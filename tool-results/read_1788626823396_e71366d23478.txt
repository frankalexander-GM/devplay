   401→   401→  if (isLoading) {

   433→   433→function AnimatedCounter({ value }: { value: number }) {
   434→   434→  const [display, setDisplay] = useState(0)
   435→   435→  useEffect(() => {
   436→   436→    const duration = 800
   437→   437→    const start = Date.now()
   438→   438→    const interval = setInterval(() => {
   439→   439→      const elapsed = Date.now() - start
   440→   440→      const progress = Math.min(elapsed / duration, 1)
   441→   441→      setDisplay(Math.floor(value * progress))
   442→   442→      if (progress >= 1) clearInterval(interval)
   443→   443→    }, 16)
   444→   444→    return () => clearInterval(interval)
   445→   445→  }, [value])
   446→   446→  return <div className="text-lg font-bold">{display}</div>
   447→   447→}
