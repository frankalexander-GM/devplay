    51→  const h = new Date().getHours()
    52→  if (h >= 6 && h < 12) return '¡Buenos días! ☕ ¿Lista la inspiración para crear hoy?'
    53→  if (h >= 12 && h < 19) return '¡Buenas tardes! 🎮 Tómate un respiro, la creatividad también descansa'

   146→export function PixelBuddy() {
   147→  const { user, isAuthed, isGuest } = useCurrentUser()
   148→  const currentView = useUIStore((s) => s.currentView)
   149→  const profileUserId = useUIStore((s) => s.profileUserId)
   150→  const openAuth = useUIStore((s) => s.openAuth)
   151→
   152→  const [collapsed, setCollapsed] = useState(false)
   153→  const [chatOpen, setChatOpen] = useState(false)
   154→  const [registerOpen, setRegisterOpen] = useState(false)
   155→  const [menuOpen, setMenuOpen] = useState(false)
   156→  const [muted, setMuted] = useState(false)
   157→  const [bubbleVisible, setBubbleVisible] = useState(true)
   158→  const [tip, setTip] = useState('¡Hola! Soy Pixel, tu ayudante de DevPlay 🤖')
   159→  const [pos, setPos] = useState({ x: 40, y: 24 })
   160→  const [facing, setFacing] = useState(1)
   161→  const [mounted, setMounted] = useState(false)
   162→  const [vp, setVp] = useState({ w: 1280, h: 800 })
   163→  const [mood, setMood] = useState<Mood>('idle')
   164→  const [emote, setEmote] = useState<{ kind: Emote; key: number }>({ kind: 'none', key: 0 })
   165→  const [particles, setParticles] = useState<{ id: number; emoji: string; dx: number }[]>([])
   166→
   167→  // Caminata 🚶, arrastre 🤏 y caiditas 🍂
   168→  const [walking, setWalking] = useState(false)
   169→  const [walkDur, setWalkDur] = useState(2)
   170→  const [dragging, setDragging] = useState(false)
   171→  const [falling, setFalling] = useState(false)
   172→  const [fallDur, setFallDur] = useState(0.4)
   173→  // Gestos al hablar 🗣️: boca y bracitos se mueven mientras Pixel responde
   174→  const [talking, setTalking] = useState(false)
   175→  const talkTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   176→  const nudgeTalk = useCallback((ms = 2400) => {
   177→    setTalking(true)
   178→    if (talkTimer.current) clearTimeout(talkTimer.current)
   179→    talkTimer.current = setTimeout(() => setTalking(false), ms)
   180→  }, [])
   181→  const dragRef = useRef<{ startX: number; startY: number; offX: number; offY: number; moved: boolean } | null>(null)
   182→  const wasDraggedRef = useRef(false)
   183→  const btnRef = useRef<HTMLButtonElement>(null)
   184→
   185→  const rootRef = btnRef
   186→  const menuRef = useRef<HTMLDivElement>(null)
   187→  const isMobile = vp.w < 1024
   188→  const mascotSize = isMobile ? 46 : 54
   189→  const canChat = isAuthed // solo devs registrados; invitados/anónimos solo ven tips
   190→
   191→  const clamp = useCallback((v: number, min: number, max: number) => Math.min(Math.max(v, min), Math.max(min, max)), [])
   192→
   193→  // ===== Posición + límites: TODA la pantalla, de pared a pared 🌍 =====
   194→  const limits = useCallback(
   195→    (w?: number) => {
   196→      const vw = w ?? vp.w
   197→      return { minX: 12, maxX: Math.max(60, vw - mascotSize - 12) }
   198→    },
   199→    [vp.w, mascotSize]
   200→  )
   201→
   202→  // ===== Viewport (resize) =====
   203→  useEffect(() => {
   204→    setMounted(true)
   205→    const w = window.innerWidth
   206→    const h = window.innerHeight
   207→    setVp({ w, h })
   208→    // Posición guardada (solo X: Pixel siempre vive en el suelo)
   209→    let initial = { x: 40, y: FLOOR_Y }
   210→    try {
   211→      const raw = localStorage.getItem('pixel-pos')
   212→      if (raw) {
   213→        const saved = JSON.parse(raw)
   214→        const { minX, maxX } = limits(w)
   215→        if (saved && typeof saved.x === 'number' && saved.x >= minX - 10 && saved.x <= maxX + 10) {
   216→          initial = { x: saved.x, y: FLOOR_Y }
   217→        }
   218→      }
   219→    } catch {}
   220→    setPos(initial)
   221→    try { setMuted(localStorage.getItem('pixel-tips-muted') === '1') } catch {}
   222→    const onR = () => setVp({ w: window.innerWidth, h: window.innerHeight })
   223→    window.addEventListener('resize', onR)
   224→    return () => window.removeEventListener('resize', onR)
   225→  }, [])
   226→
   227→  // Mantener dentro de la pantalla al redimensionar
   228→  useEffect(() => {
   229→    if (!mounted) return
   230→    setPos((p) => {
   231→      const { maxX } = limits()
   232→      if (p.x > maxX) return { ...p, x: maxX }
   233→      return p
   234→    })
   235→  }, [vp.w, mascotSize, mounted, limits])
   236→
   237→  const savePos = useCallback((p: { x: number; y: number }) => {
   238→    try { localStorage.setItem('pixel-pos', JSON.stringify(p)) } catch {}
   239→  }, [])
   240→
   241→  // ===== Caminar por todo el suelo, de pared a pared 🚶 =====
   242→  useEffect(() => {
   243→    if (!mounted) return
   244→    const interval = setInterval(() => {
   245→      if (chatOpen || registerOpen || dragging || falling) return
   246→      const { minX, maxX } = limits()
   247→      const span = maxX - minX
   248→      // A veces paseo cortito, a veces atravieso TODA la pantalla
   249→      const x = Math.random() < 0.45
   250→        ? (Math.random() < 0.5 ? minX + Math.random() * span * 0.3 : maxX - Math.random() * span * 0.3)
   251→        : minX + Math.random() * span
   252→      setPos((prev) => {
   253→        if (Math.abs(x - prev.x) < 50) return prev
   254→        setFacing(x > prev.x ? 1 : -1)
   255→        setWalkDur(Math.min(Math.max(Math.abs(x - prev.x) / WALK_SPEED, 0.8), 9))
   256→        setWalking(true)
   257→        return { x, y: FLOOR_Y }
   258→      })
   259→    }, 15000)
   260→    return () => clearInterval(interval)
   261→  }, [mounted, chatOpen, registerOpen, dragging, falling, limits])
   262→
   263→  // ===== Personalidad: emotes (saltar, bailar, guiñar, chispas) =====
   264→  const doEmote = useCallback((kind: Emote, m: Mood = 'idle') => {
   265→    setEmote((e) => ({ kind, key: e.key + 1 }))
   266→    if (m !== 'idle') {
   267→      setMood(m)
   268→      setTimeout(() => setMood('idle'), 2000)
   269→    }
   270→  }, [])
   271→
   272→  const spawnParticles = useCallback((n = 1) => {
   273→    const batch = Array.from({ length: n }, (_, i) => ({
   274→      id: Date.now() + i + Math.random(),
   275→      emoji: PARTICLE_EMOJIS[Math.floor(Math.random() * PARTICLE_EMOJIS.length)],
   276→      dx: Math.random() * 48 - 16,
   277→    }))
   278→    setParticles((p) => [...p.slice(-6), ...batch])
   279→    setTimeout(() => setParticles((p) => p.filter((x) => !batch.some((b) => b.id === x.id))), 1600)
   280→  }, [])
   281→
   282→  useEffect(() => {
   283→    if (!mounted || collapsed || chatOpen || registerOpen || walking || dragging || falling || mood === 'dizzy') return
   284→    const t = setInterval(() => {
   285→      const r = Math.random()
   286→      if (r < 0.3) doEmote('jump', 'happy')
   287→      else if (r < 0.55) doEmote('dance')
   288→      else if (r < 0.8) { doEmote('jump', 'excited'); spawnParticles(2) }
   289→      else doEmote('none', 'wink')
   290→    }, 15000)
   291→    return () => clearInterval(t)
   292→  }, [mounted, collapsed, chatOpen, registerOpen, walking, dragging, falling, mood, doEmote, spawnParticles])
   293→
   294→  // ===== Dormilón: si nadie lo pela en 100s, se duerme 💤 =====
   295→  const idleTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   296→  const resetIdle = useCallback(() => {
   297→    setMood((m) => (m === 'sleepy' ? 'idle' : m))
   298→    if (idleTimer.current) clearTimeout(idleTimer.current)
   299→    idleTimer.current = setTimeout(() => {
   300→      setMood('sleepy')
   301→    }, 100000)
   302→  }, [])
   303→  useEffect(() => {
   304→    if (!mounted || collapsed) return
   305→    resetIdle()
   306→    return () => { if (idleTimer.current) clearTimeout(idleTimer.current) }
   307→  }, [mounted, collapsed, resetIdle])
   308→
   309→  // ===== Tips contextuales =====
   310→  const isOwnProfile = currentView === 'profile' && profileUserId && user?.id === profileUserId
   311→  const profileTarget = isOwnProfile ? user?.id : currentView === 'profile' ? profileUserId : null
   312→
   313→  const { data: profileData } = useQuery({
   314→    queryKey: ['user-profile', profileTarget],
   315→    queryFn: () => userService.get(profileTarget!),
   316→    enabled: currentView === 'profile' && !!profileTarget,
   317→    staleTime: 30_000,
   318→  })
   319→
   320→  useEffect(() => {
   321→    if (collapsed || muted) {
   322→      setBubbleVisible(false)
   323→      return
   324→    }
   325→    const noPostsYet = Boolean(isOwnProfile) && (profileData?.posts?.length ?? 0) === 0
   326→    const tips =
   327→      mood === 'sleepy'
   328→        ? ['Zzz... 💤 (tócame para despertarme)']
   329→        : tipsForView(currentView, { isOwnProfile: !!isOwnProfile, noPostsYet, canChat })
   330→    if (!isAuthed) tips.push('Truco: crea una cuenta gratis para dar like, seguir devs y publicar 😉')
   331→    let i = 0
   332→    setBubbleVisible(true)
   333→    setTip(tips[0])
   334→    const rotate = setInterval(() => {
   335→      i = (i + 1) % tips.length
   336→      setBubbleVisible(false)
   337→      setTimeout(() => {
   338→        setTip(tips[i])
   339→        setBubbleVisible(true)
   340→      }, 350)
   341→    }, 12000)
   342→    return () => clearInterval(rotate)
   343→  }, [currentView, isOwnProfile, profileData, isAuthed, collapsed, muted, mood, canChat])
   344→
   345→  // ===== Ojos que siguen el cursor =====
   346→  const pupilX = useSpring(useMotionValue(0), { stiffness: 300, damping: 22 })
   347→  const pupilY = useSpring(useMotionValue(0), { stiffness: 300, damping: 22 })
   348→
   349→  useEffect(() => {
   350→    function onMouseMove(e: MouseEvent) {
   351→      const el = rootRef.current
   352→      if (!el) return
   353→      const rect = el.getBoundingClientRect()
   354→      const cx = rect.left + rect.width / 2
   355→      const cy = rect.top + rect.height / 2
   356→      const dx = e.clientX - cx
   357→      const dy = e.clientY - cy
   358→      const dist = Math.hypot(dx, dy) || 1
   359→      const strength = Math.min(dist / 220, 1)
   360→      pupilX.set((dx / dist) * 2.4 * strength)
   361→      pupilY.set((dy / dist) * 2 * strength)
   362→    }
   363→    window.addEventListener('mousemove', onMouseMove, { passive: true })
   364→    return () => window.removeEventListener('mousemove', onMouseMove)
   365→  }, [pupilX, pupilY])
   366→
   367→  // ===== Click y arrastre 🤏 =====
   368→  function actualPos(): { x: number; y: number } {
   369→    const el = btnRef.current
   370→    if (!el) return pos
   371→    const r = el.getBoundingClientRect()
   372→    return { x: r.left, y: window.innerHeight - r.bottom }
   373→  }

   375→  // ===== Caída con gravedad 🍂: al soltar, cae hasta el suelo con carita de susto =====
   376→  function startFall() {
   377→    const p = actualPos()
   378→    const { minX, maxX } = limits()
   379→    const x = clamp(p.x, minX - 2, maxX + 2)
   380→    const h = Math.max(0, p.y - FLOOR_Y)
   381→    if (h < 4) {
   382→      // Ya estaba en el suelo: solo un plop de aterrizaje
   383→      setPos((pp) => { savePos({ x: pp.x, y: FLOOR_Y }); return { x: pp.x, y: FLOOR_Y } })
   384→      doEmote('land')
   385→      spawnParticles(2)
   386→      setMood('dizzy')
   387→      setTimeout(() => setMood('happy'), 1000)
   388→      setTimeout(() => setMood('idle'), 2400)
   389→      return
   390→    }
   391→    setFallDur(Math.min(Math.max(Math.sqrt(h / 2400), 0.16), 0.75))
   392→    setMood('shocked')
   393→    setFalling(true)
   394→    setPos({ x, y: FLOOR_Y })
   395→  }

   397→  function landIt() {
   398→    setFalling(false)
   399→    setPos((p) => { savePos({ x: p.x, y: FLOOR_Y }); return p })
   400→    doEmote('land')
   401→    spawnParticles(2)
   402→    setMood('dizzy')
   403→    setTimeout(() => setMood('happy'), 1000)
   404→    setTimeout(() => setMood('idle'), 2400)
   405→  }

   407→  function handleBuddyClick() {
   408→    resetIdle()
   409→    setBubbleVisible(false)
   410→    if (mood === 'sleepy') {
   411→      setMood('excited')
   412→      setTimeout(() => setMood('idle'), 1500)
   413→      spawnParticles(2)
   414→      setTip('¡Aaah! Me quedé dormido 😴 ¿En qué andamos?')
   415→      setBubbleVisible(true)
   416→      setTimeout(() => setBubbleVisible(false), 4200)
   417→      return
   418→    }
   419→    doEmote('jump', 'happy')
   420→    spawnParticles(1)
   421→    if (canChat) {
   422→      setRegisterOpen(false)
   423→      setChatOpen((v) => !v)
   424→    } else {
   425→      setChatOpen(false)
   426→      setRegisterOpen((v) => !v)
   427→    }
   428→  }

   430→  function onPointerDown(e: React.PointerEvent) {
   431→    if (collapsed) return
   432→    try { (e.currentTarget as Element).setPointerCapture?.(e.pointerId) } catch {}
   433→    const p = actualPos()
   434→    // Detener caminata donde esté y fijar posición real
   435→    setWalking(false)
   436→    setPos(p)
   437→    dragRef.current = { startX: e.clientX, startY: e.clientY, offX: e.clientX - p.x, offY: (window.innerHeight - e.clientY) - p.y, moved: false }
   438→  }
   439→
   440→  function onPointerMove(e: React.PointerEvent) {
   441→    const d = dragRef.current
   442→    if (!d) return
   443→    const dx = e.clientX - d.startX
   444→    const dy = (window.innerHeight - e.clientY) - (window.innerHeight - d.startY)
   445→    if (!d.moved && Math.hypot(dx, dy) > 7) {
   446→      d.moved = true
   447→      wasDraggedRef.current = true
   448→      setDragging(true)
   449→      setBubbleVisible(false)
   450→      setChatOpen(false)
   451→      setRegisterOpen(false)
   452→      setMood('excited')
   453→    }
   454→    if (!d.moved) return
   455→    const { minX, maxX } = limits()
   456→    setFacing(dx >= 0 ? 1 : -1)
   457→    setPos({
   458→      x: clamp(e.clientX - d.offX, minX - 6, maxX + 10),
   459→      y: clamp((window.innerHeight - e.clientY) - d.offY, FLOOR_Y, Math.max(FLOOR_Y, vp.h - mascotSize - 14)),
   460→    })
   461→  }
   462→
   463→  function onPointerUp() {
   464→    const d = dragRef.current
   465→    dragRef.current = null
   466→    if (!d) return
   467→    if (d.moved) {
   468→      wasDraggedRef.current = false
   469→      setDragging(false)
   470→      startFall()
   471→    } else {
   472→      handleBuddyClick()
   473→    }
