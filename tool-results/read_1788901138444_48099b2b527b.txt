   177→    queryKey: ['store-my-items'],

   183→

   185→  const buyMutation = useMutation({
   186→    mutationFn: (itemId: string) => storeService.buy(itemId),
   187→    onSuccess: (data) => {

   189→        description: `-${data.purchase.pricePaid} DevCoins · Balance: ${data.balance}`,

   192→      qc.invalidateQueries({ queryKey: ['store-balance'] })

   209→  const transactions: DevCoinTransaction[] = balanceData?.transactions ?? []

   257→              { n: 'I', icon: Rocket, title: 'Gana', text: 'Publica devlogs, ayuda en betas y recibe likes. Cada aporte suma monedas.' },

   267→                    {step.n}
   268→                  </span>
   269→                </div>
   270→                <p className="font-display font-bold text-sm mb-1">{step.title}</p>
   271→                <p className="text-xs text-muted-foreground leading-relaxed">{step.text}</p>
   272→              </div>
   273→            ))}
   274→          </div>
   275→          <div className="rule-ornate w-40 mx-auto mt-5 opacity-60">
   276→            <span className="text-[9px]">◆</span>
   277→          </div>
   278→          <p className="text-center text-xs text-muted-foreground italic mt-4">
   279→            Los DevCoins no se compran con dinero real — se ganan participando.

   322→            key={cat}

   325→            icon={CATEGORY_META[cat].icon}

   388→            <StoreCard
   389→              key={item.id}
   390→              item={item}
   391→              owned={ownedItemIds.has(item.id)}
   392→              canAfford={balance >= item.price}
   393→              disabled={isGuest}
   394→              comingSoon={STORE_CLOSED}
   395→              buying={buyMutation.isPending && buyMutation.variables === item.id}
   396→              onBuy={() => buyMutation.mutate(item.id)}
   397→              index={i}
   398→            />
   399→          ))}

   403→      {/* ===== Mis artículos ===== */}

   415→        {myItemsLoading ? (

   417→            {[1, 2, 3].map((i) => (
   418→              <Skeleton key={i} className="h-24 rounded-md" />
   419→            ))}
   420→          </div>
   421→        ) : myItems.length === 0 ? (
   422→          <div className="glass-card p-8 text-center">
   423→            <Gift className="mx-auto mb-2 h-10 w-10 text-muted-foreground/60" />
   424→            <p className="font-semibold">Aún no tienes artículos</p>
   425→            <p className="text-sm text-muted-foreground mt-1">

   427→            </p>
   428→          </div>

   430→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   431→            {myItems.map((p, i) => (
   432→              <OwnedItemCard key={p.id} purchase={p} index={i} />
   433→            ))}
   434→          </div>
   435→        )}
   436→      </section>
   437→
   438→      {/* ===== Historial de transacciones (colapsible) ===== */}
   439→      <section className="pt-2">
   440→        <Collapsible open={txHistoryOpen} onOpenChange={setTxHistoryOpen}>
   441→          <CollapsibleTrigger asChild>
   442→            <button className="flex w-full items-center justify-between rounded-md bg-secondary/40 px-4 py-3 hover:bg-secondary/60 transition">
   443→              <span className="flex items-center gap-2 font-semibold">
   444→                <History className="h-4 w-4 text-primary" />
   445→                Historial de transacciones
   446→                {transactions.length > 0 && (
   447→                  <Badge variant="secondary" className="rounded-full">
   448→                    {transactions.length}
   449→                  </Badge>
   450→                )}
   451→              </span>
   452→              <ChevronDown
   453→                className={cn(

   456→                )}
   457→              />
   458→            </button>
   459→          </CollapsibleTrigger>
   460→          <CollapsibleContent>

   464→                  No hay transacciones todavía
   465→                </div>

   467→                transactions.map((tx) => (
   468→                  <TransactionRow key={tx.id} tx={tx} />

   470→              )}
   471→            </div>
   472→          </CollapsibleContent>
   473→        </Collapsible>
   474→      </section>
   475→
   476→      <div className="h-4" />
   477→    </div>
   478→  )
   479→}
