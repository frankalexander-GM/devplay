     1→'use client'
     2→
     3→import { useState, useEffect } from 'react'
     4→import { signIn } from 'next-auth/react'
     5→import { motion, AnimatePresence } from 'framer-motion'
     6→import { Dialog, DialogContent, DialogTitle } from '@/components/ui/dialog'
     7→import { Button } from '@/components/ui/button'
     8→import { Input } from '@/components/ui/input'
     9→import { Label } from '@/components/ui/label'
    10→import { Checkbox } from '@/components/ui/checkbox'
    11→import { useUIStore } from '@/lib/stores'
    12→import { useCurrentUser } from '@/hooks/use-current-user'
    13→import { authService, userService } from '@/services/devplay-service'
    14→import { toast } from 'sonner'
    15→import {
    16→  Eye, EyeOff, Loader2, Mail, Lock, AtSign, Shield,
    17→  Check, AlertCircle, Users, ArrowRight, ArrowBigUp,
    18→} from 'lucide-react'
    19→import { cn } from '@/lib/utils'
    20→import { ForgotPasswordModal } from '@/components/devplay/forgot-password-modal'
    21→
    22→type FieldError = { field: string; message: string }
    23→
    24→export function AuthModal() {
    25→  const { authModalOpen, authMode, closeAuth, setView } = useUIStore()
    26→  const { refreshAfterLogin, loginAsGuest } = useCurrentUser()
    27→  const [activeTab, setActiveTab] = useState<'login' | 'register'>(authMode)
    28→
    29→  useEffect(() => {
    30→    if (authModalOpen) setActiveTab(authMode)
    31→  }, [authModalOpen, authMode])
    32→
    33→  const [loading, setLoading] = useState(false)
    34→  const [guestLoading, setGuestLoading] = useState(false)
    35→  const [googleLoading, setGoogleLoading] = useState(false)
    36→  // Botón de Google SOLO cuando las llaves OAuth estén configuradas en el
    37→  // servidor real (NEXT_PUBLIC_GOOGLE_ENABLED=1). Mientras tanto, oculto para
    38→  // no mostrar un botón roto.
    39→  const googleEnabled = process.env.NEXT_PUBLIC_GOOGLE_ENABLED === '1'
    40→  const [errors, setErrors] = useState<FieldError[]>([])
    41→  const [showForgot, setShowForgot] = useState(false)
    42→
    43→  // login state
    44→  const [loginEmail, setLoginEmail] = useState('')
    45→  const [loginPassword, setLoginPassword] = useState('')
    46→  const [showLoginPass, setShowLoginPass] = useState(false)
    47→  const [rememberMe, setRememberMe] = useState(true)
    48→  const [capsLockOn, setCapsLockOn] = useState(false)
    49→
    50→  // register state
    51→  const [regEmail, setRegEmail] = useState('')
    52→  const [regUsername, setRegUsername] = useState('')
    53→  const [regPassword, setRegPassword] = useState('')
    54→  const [showRegPass, setShowRegPass] = useState(false)
    55→  const [agreeTerms, setAgreeTerms] = useState(false)
    56→  const [usernameTaken, setUsernameTaken] = useState(false)
    57→  const [checkingUsername, setCheckingUsername] = useState(false)
    58→
    59→  // ===== Validaciones en tiempo real =====
    60→  const emailValid = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(loginEmail || regEmail)
    61→  const usernameValid = /^[a-zA-Z0-9_]{3,20}$/.test(regUsername)
    62→  const usernameFree = usernameValid && !checkingUsername && !usernameTaken // disponible de verdad
    63→  const passwordStrength = getPasswordStrength(regPassword)
    64→
    65→  // ===== Disponibilidad real del nombre de usuario (con espera) =====
    66→  useEffect(() => {
    67→    setUsernameTaken(false)
    68→    if (!usernameValid) {
    69→      setCheckingUsername(false)
    70→      return
    71→    }
    72→    setCheckingUsername(true)
    73→    const t = setTimeout(async () => {
    74→      try {
    75→        const res = await userService.getByUsername(regUsername)
    76→        setUsernameTaken(!!res?.user)
    77→      } catch {
    78→        setUsernameTaken(false)
    79→      } finally {
    80→        setCheckingUsername(false)
    81→      }
    82→    }, 500)
    83→    return () => clearTimeout(t)
    84→  }, [regUsername, usernameValid])
    85→
    86→  // Sugerencia amable: proponer usuario a partir del correo
    87→  function suggestUsername() {
    88→    if (!regUsername && regEmail.includes('@')) {
    89→      const base = regEmail.split('@')[0].replace(/[^a-zA-Z0-9_]/g, '').slice(0, 16)
    90→      if (base.length >= 3) setRegUsername(base)
    91→    }
    92→  }
    93→
    94→  // (validaciones definidas arriba)
    95→
    96→  function getPasswordStrength(pwd: string): { score: number; label: string; color: string } {
    97→    let score = 0
    98→    if (pwd.length >= 6) score++
    99→    if (pwd.length >= 10) score++
   100→    if (/[A-Z]/.test(pwd) && /[a-z]/.test(pwd)) score++
   101→    if (/[0-9]/.test(pwd)) score++
   102→    if (/[^A-Za-z0-9]/.test(pwd)) score++
   103→    const labels = ['Muy débil', 'Débil', 'Regular', 'Buena', 'Fuerte', 'Muy fuerte']
   104→    const colors = ['bg-red-500', 'bg-bronze-500', 'bg-amber-500', 'bg-lime-500', 'bg-olive-500', 'bg-olive-600']
   105→    return { score, label: labels[score] || 'Muy débil', color: colors[score] || 'bg-red-500' }
   106→  }
   107→
   108→  function getFieldError(field: string): string | undefined {
   109→    return errors.find(e => e.field === field)?.message
   110→  }
   111→
   112→  function clearFieldError(field: string) {
   113→    setErrors(prev => prev.filter(e => e.field !== field))
   114→  }
   115→
   116→  function detectCapsLock(e: React.KeyboardEvent) {
   117→    setCapsLockOn(!!e.getModifierState('CapsLock'))
   118→  }
   119→
   120→  async function handleLogin(e: React.FormEvent) {
   121→    e.preventDefault()
   122→    const newErrors: FieldError[] = []
   123→    if (!emailValid) newErrors.push({ field: 'loginEmail', message: 'Email inválido' })
   124→    if (loginPassword.length < 1) newErrors.push({ field: 'loginPassword', message: 'Contraseña requerida' })
   125→    if (newErrors.length) { setErrors(newErrors); return }
   126→
   127→    setLoading(true)
   128→    setErrors([])
   129→    try {
   130→      const res = await signIn('credentials', {
   131→        email: loginEmail,
   132→        password: loginPassword,
   133→        redirect: false,
   134→      })
   135→      if (res?.error) {
   136→        setErrors([{ field: 'loginPassword', message: 'Email o contraseña incorrectos' }])
   137→      } else {
   138→        toast.success('¡Bienvenido de vuelta!')
   139→        setLoginEmail('')
   140→        setLoginPassword('')
   141→        await refreshAfterLogin()
   142→        closeAuth()
   143→      }
   144→    } catch {
   145→      setErrors([{ field: 'loginEmail', message: 'Error de conexión. Intenta de nuevo.' }])
   146→    } finally {
   147→      setLoading(false)
   148→    }
   149→  }
   150→
   151→  async function handleRegister(e: React.FormEvent) {
   152→    e.preventDefault()
   153→    const newErrors: FieldError[] = []
   154→    if (!emailValid) newErrors.push({ field: 'regEmail', message: 'Email inválido' })
   155→    if (!usernameValid) newErrors.push({ field: 'regUsername', message: '3-20 caracteres: letras, números y _' })
   156→    else if (usernameTaken) newErrors.push({ field: 'regUsername', message: 'Ese nombre ya está en uso, prueba otro' })
   157→    if (regPassword.length < 6) newErrors.push({ field: 'regPassword', message: 'Mínimo 6 caracteres' })
   158→    if (!agreeTerms) newErrors.push({ field: 'terms', message: 'Debes aceptar los términos' })
   159→    if (newErrors.length) { setErrors(newErrors); return }
   160→
   161→    setLoading(true)
   162→    setErrors([])
   163→    try {
   164→      await authService.register({ email: regEmail, username: regUsername, password: regPassword })
   165→      await signIn('credentials', { email: regEmail, password: regPassword, redirect: false })
   166→      toast.success('¡Cuenta creada! Bienvenido a DevPlay')
   167→      setRegEmail(''); setRegUsername(''); setRegPassword(''); setAgreeTerms(false)
   168→      await refreshAfterLogin()
   169→      closeAuth()
   170→    } catch (err: any) {
   171→      setErrors([{ field: 'regEmail', message: err.message || 'Error al registrarse' }])
   172→    } finally {
   173→      setLoading(false)
   174→    }
   175→  }
   176→
   177→  async function handleGuest() {
   178→    setGuestLoading(true)
   179→    try {
   180→      await loginAsGuest()
   181→      toast.success('Explorando como invitado')
   182→      closeAuth()
   183→    } catch {
   184→      toast.error('Error al entrar como invitado')
   185→    } finally {
   186→      setGuestLoading(false)
   187→    }
   188→  }
   189→
   190→  return (
   191→    <Dialog open={authModalOpen} onOpenChange={(o) => !o && closeAuth()}>
   192→      <DialogContent className="glass-strong max-w-md rounded-lg p-0 overflow-hidden border-border/60">
   193→        {/* Título oculto: solo para lectores de pantalla */}
   194→        <DialogTitle className="sr-only">{activeTab === 'login' ? 'Iniciar sesión' : 'Crear cuenta'}</DialogTitle>
   195→        {/* ===== Header con branding ===== */}
   196→        <div className="relative px-6 pt-6 pb-4 text-center border-b border-border/40">
   197→          <div className="absolute inset-0 bg-gradient-to-b from-primary/10 via-transparent to-transparent" />
   198→          <div className="relative">
   199→            <div className="mx-auto mb-3 flex h-14 w-14 items-center justify-center rounded-lg bg-gradient-to-br from-primary to-accent shadow-lg">
   200→              <img src="/logo-devplay.png" alt="DevPlay" className="h-10 w-10 rounded-md object-cover" />
   201→            </div>
   202→            <h2 className="font-display text-3xl font-black tracking-tight text-foreground">DevPlay</h2>
   203→            <p className="label-caps mt-1">
   204→              Comunidad para devs de videojuegos indie
   205→            </p>
   206→          </div>
   207→        </div>
   208→
   209→        {/* ===== Tab switcher ===== */}
   210→        <div className="px-6 pt-4">
   211→          <div className="grid w-full grid-cols-2 gap-1 rounded-lg bg-secondary/50 p-1">
   212→            <button
   213→              type="button"
   214→              onClick={() => { setActiveTab('login'); setErrors([]) }}
   215→              className={cn(
   216→                'rounded-md py-2 text-sm font-semibold transition-all',
   217→                activeTab === 'login'
   218→                  ? 'btn-gradient-primary shadow-sm'
   219→                  : 'text-muted-foreground hover:text-foreground'
   220→              )}
   221→            >
   222→              Iniciar sesión
   223→            </button>
   224→            <button
   225→              type="button"
   226→              onClick={() => { setActiveTab('register'); setErrors([]) }}
   227→              className={cn(
   228→                'rounded-md py-2 text-sm font-semibold transition-all',
   229→                activeTab === 'register'
   230→                  ? 'btn-gradient-primary shadow-sm'
   231→                  : 'text-muted-foreground hover:text-foreground'
   232→              )}
   233→            >
   234→              Crear cuenta
   235→            </button>
   236→          </div>
   237→        </div>
   238→
   239→        {/* ===== Formularios ===== */}
   240→        <div className="px-6 pb-6 pt-4">
   241→          <AnimatePresence mode="wait">
   242→            {activeTab === 'login' ? (
   243→              <motion.form
   244→                key="login"
   245→                initial={{ opacity: 0, x: -10 }}
   246→                animate={{ opacity: 1, x: 0 }}
   247→                exit={{ opacity: 0, x: 10 }}
   248→                transition={{ duration: 0.2 }}
   249→                onSubmit={handleLogin}
   250→                className="space-y-4"
   251→              >
   252→                {/* Email */}
   253→                <FormField
   254→                  label="Email"
   255→                  icon={Mail}
   256→                  error={getFieldError('loginEmail')}
   257→                >
   258→                  <Input
   259→                    type="email"
   260→                    value={loginEmail}
   261→                    onChange={(e) => { setLoginEmail(e.target.value); clearFieldError('loginEmail') }}
   262→                    placeholder="tu@email.com"
   263→                    required
   264→                    autoComplete="email"
   265→                    className={cn(
   266→                      'rounded-md pl-10 pr-4 h-11',
   267→                      getFieldError('loginEmail') && 'border-red-500/50 focus-visible:ring-red-500/30'
   268→                    )}
   269→                  />
   270→                </FormField>
   271→
   272→                {/* Password */}
   273→                <FormField
   274→                  label="Contraseña"
   275→                  icon={Lock}
   276→                  error={getFieldError('loginPassword')}
   277→                  trailing={
   278→                    <button
   279→                      type="button"
   280→                      onClick={() => setShowLoginPass(!showLoginPass)}
   281→                      className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground transition"
   282→                    >
   283→                      {showLoginPass ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   284→                    </button>
   285→                  }
   286→                >
   287→                  <Input
   288→                    type={showLoginPass ? 'text' : 'password'}
   289→                    value={loginPassword}
   290→                    onChange={(e) => { setLoginPassword(e.target.value); clearFieldError('loginPassword') }}
   291→                    onKeyDown={detectCapsLock}
   292→                    placeholder="••••••••"
   293→                    required
   294→                    autoComplete="current-password"
   295→                    className={cn(
   296→                      'rounded-md pl-10 pr-10 h-11',
   297→                      getFieldError('loginPassword') && 'border-red-500/50 focus-visible:ring-red-500/30'
   298→                    )}
   299→                  />
   300→                  {capsLockOn && (
   301→                    <p className="text-[10px] text-amber-500 mt-1 flex items-center gap-1">
   302→                      <ArrowBigUp className="h-3 w-3" /> Bloq Mayús activado
   303→                    </p>
   304→                  )}
   305→                </FormField>
   306→
   307→                {/* Remember + Forgot */}
   308→                <div className="flex items-center justify-between text-xs">
   309→                  <label className="flex items-center gap-2 cursor-pointer">
   310→                    <Checkbox checked={rememberMe} onCheckedChange={(v) => setRememberMe(!!v)} />
   311→                    <span className="text-muted-foreground">Recuérdame</span>
   312→                  </label>
   313→                  <button
   314→                    type="button"
   315→                    onClick={() => setShowForgot(true)}
   316→                    className="text-primary hover:underline font-medium"
   317→                  >
   318→                    ¿Olvidaste tu contraseña?
   319→                  </button>
   320→                </div>
   321→
   322→                {/* Submit */}
   323→                <Button
   324→                  type="submit"
   325→                  disabled={loading}
   326→                  className="w-full btn-gradient-primary rounded-md h-11 font-semibold"
   327→                >
   328→                  {loading ? (
   329→                    <><Loader2 className="mr-2 h-4 w-4 animate-spin" /> Entrando...</>
   330→                  ) : (
   331→                    <>Entrar <ArrowRight className="ml-2 h-4 w-4" /></>
   332→                  )}
   333→                </Button>
   334→              </motion.form>
   335→            ) : (
   336→              <motion.form
   337→                key="register"
   338→                initial={{ opacity: 0, x: 10 }}
   339→                animate={{ opacity: 1, x: 0 }}
   340→                exit={{ opacity: 0, x: -10 }}
   341→                transition={{ duration: 0.2 }}
   342→                onSubmit={handleRegister}
   343→                className="space-y-4"
   344→              >
   345→                {/* Beneficios */}
   346→                <div className="flex items-center justify-center gap-3 text-[10px] font-medium text-muted-foreground bg-secondary/40 rounded-md py-2 px-3">
   347→                  <span className="flex items-center gap-1"><Check className="h-3 w-3 text-olive-500" /> Gratis</span>
   348→                  <span className="flex items-center gap-1"><Check className="h-3 w-3 text-olive-500" /> Publica y comparte</span>
   349→                  <span className="flex items-center gap-1"><Check className="h-3 w-3 text-olive-500" /> Habla con Pixel 🤖</span>
   350→                </div>
   351→
   352→                {/* Email */}
   353→                <FormField
   354→                  label="Email"
   355→                  icon={Mail}
   356→                  error={getFieldError('regEmail')}
   357→                  success={emailValid && regEmail.length > 0 ? 'Email válido' : undefined}
   358→                >
   359→                  <Input
   360→                    type="email"
   361→                    value={regEmail}
   362→                    onChange={(e) => { setRegEmail(e.target.value); clearFieldError('regEmail') }}
   363→                    placeholder="tu@email.com"
   364→                    required
   365→                    autoComplete="email"
   366→                    className={cn(
   367→                      'rounded-md pl-10 pr-10 h-11',
   368→                      getFieldError('regEmail') && 'border-red-500/50 focus-visible:ring-red-500/30',
   369→                      emailValid && regEmail.length > 0 && 'border-olive-500/50 focus-visible:ring-olive-500/30'
   370→                    )}
   371→                  />
   372→                  {emailValid && regEmail.length > 0 && (
   373→                    <Check className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 text-olive-500" />
   374→                  )}
   375→                </FormField>
   376→
   377→                {/* Username */}
   378→                <FormField
   379→                  label="Nombre de usuario"
   380→                  icon={AtSign}
   381→                  error={getFieldError('regUsername') || (usernameTaken ? 'Ya en uso, prueba otro' : undefined)}
   382→                  success={usernameFree ? 'Disponible' : undefined}
   383→                >
   384→                  <Input
   385→                    value={regUsername}
   386→                    onChange={(e) => { setRegUsername(e.target.value); clearFieldError('regUsername') }}
   387→                    onBlur={suggestUsername}
   388→                    placeholder="SuperCreador99"
   389→                    pattern="[a-zA-Z0-9_]{3,20}"
   390→                    required
   391→                    autoComplete="username"
   392→                    className={cn(
   393→                      'rounded-md pl-10 pr-10 h-11',
   394→                      (getFieldError('regUsername') || usernameTaken) && 'border-red-500/50 focus-visible:ring-red-500/30',
   395→                      usernameFree && 'border-olive-500/50 focus-visible:ring-olive-500/30'
   396→                    )}
   397→                  />
   398→                  {checkingUsername ? (
   399→                    <Loader2 className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground animate-spin" />
   400→                  ) : usernameFree ? (
   401→                    <Check className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 text-olive-500" />
   402→                  ) : null}
   403→                </FormField>
   404→
   405→                {/* Password */}
   406→                <FormField
   407→                  label="Contraseña"
   408→                  icon={Lock}
   409→                  error={getFieldError('regPassword')}
   410→                  trailing={
   411→                    <button
   412→                      type="button"
   413→                      onClick={() => setShowRegPass(!showRegPass)}
   414→                      className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground transition"
   415→                    >
   416→                      {showRegPass ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   417→                    </button>
   418→                  }
   419→                >
   420→                  <Input
   421→                    type={showRegPass ? 'text' : 'password'}
   422→                    value={regPassword}
   423→                    onChange={(e) => { setRegPassword(e.target.value); clearFieldError('regPassword') }}
   424→                    onKeyDown={detectCapsLock}
   425→                    placeholder="Mínimo 6 caracteres"
   426→                    minLength={6}
   427→                    required
   428→                    autoComplete="new-password"
   429→                    className={cn(
   430→                      'rounded-md pl-10 pr-10 h-11',
   431→                      getFieldError('regPassword') && 'border-red-500/50 focus-visible:ring-red-500/30'
   432→                    )}
   433→                  />
   434→                  {capsLockOn && (
   435→                    <p className="text-[10px] text-amber-500 mt-1 flex items-center gap-1">
   436→                      <ArrowBigUp className="h-3 w-3" /> Bloq Mayús activado
   437→                    </p>
   438→                  )}
   439→                  {/* Strength meter */}
   440→                  {regPassword.length > 0 && (
   441→                    <div className="mt-2 space-y-1">
   442→                      <div className="flex gap-1">
   443→                        {[1,2,3,4,5].map(i => (
   444→                          <div
   445→                            key={i}
   446→                            className={cn(
   447→                              'h-1 flex-1 rounded-full transition-colors',
   448→                              i <= passwordStrength.score ? passwordStrength.color : 'bg-secondary'
   449→                            )}
   450→                          />
   451→                        ))}
   452→                      </div>
   453→                      <p className="text-[10px] text-muted-foreground">
   454→                        Fortaleza: <span className="font-medium">{passwordStrength.label}</span>
   455→                      </p>
   456→                    </div>
   457→                  )}
   458→                </FormField>
   459→
   460→                {/* Terms */}
   461→                <div>
   462→                  <label className="flex items-start gap-2 cursor-pointer">
   463→                    <Checkbox
   464→                      checked={agreeTerms}
   465→                      onCheckedChange={(v) => { setAgreeTerms(!!v); clearFieldError('terms') }}
   466→                      className="mt-0.5"
   467→                    />
   468→                    <span className="text-xs text-muted-foreground leading-relaxed">
   469→                      Acepto los <button type="button" onClick={() => { closeAuth(); setView('about') }} className="text-primary hover:underline">Términos de servicio</button> y la{' '}
   470→                      <button type="button" onClick={() => { closeAuth(); setView('about') }} className="text-primary hover:underline">Política de privacidad</button>
   471→                    </span>
   472→                  </label>
   473→                  {getFieldError('terms') && (
   474→                    <p className="text-[10px] text-red-500 mt-1 flex items-center gap-1">
   475→                      <AlertCircle className="h-3 w-3" /> {getFieldError('terms')}
   476→                    </p>
   477→                  )}
   478→                </div>
   479→
   480→                {/* Submit */}
   481→                <Button
   482→                  type="submit"
   483→                  disabled={loading || !agreeTerms}
   484→                  className="w-full btn-gradient-primary rounded-md h-11 font-semibold"
   485→                >
   486→                  {loading ? (
   487→                    <><Loader2 className="mr-2 h-4 w-4 animate-spin" /> Creando...</>
   488→                  ) : (
   489→                    <>Crear cuenta <ArrowRight className="ml-2 h-4 w-4" /></>
   490→                  )}
   491→                </Button>
   492→              </motion.form>
   493→            )}
   494→          </AnimatePresence>
   495→
   496→          {/* ===== Divider + Social login (solo con OAuth configurado) ===== */}
   497→          {googleEnabled && (
   498→            <>
   499→              <div className="relative my-5">
   500→                <div className="absolute inset-0 flex items-center">
   501→                  <span className="w-full border-t border-border/50" />
   502→                </div>
   503→                <div className="relative flex justify-center text-xs uppercase">
   504→                  <span className="glass-strong px-3 text-muted-foreground rounded-sm label-caps">o continúa con</span>
   505→                </div>
   506→              </div>
   507→              <div>
   508→                <SocialButton
   509→                  provider="google"
   510→                  loading={googleLoading}
   511→                  onClick={async () => {
   512→                    setGoogleLoading(true)
   513→                    try {
   514→                      await signIn('google', { callbackUrl: '/' })
   515→                    } catch {
   516→                      toast.error('Error al conectar con Google')
